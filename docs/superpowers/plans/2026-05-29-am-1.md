> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Commit a generated OpenAPI 3.1 `openapi.json` describing the EMHASS JSON API, plus `scripts/generate_openapi.py` (stdlib, hybrid url_map+curated) and a pytest drift/validity guard.

**Architecture:** A stdlib generator imports `emhass.web_server`, reads `app.url_map` for the authoritative route set, pairs each API route with a curated schema, skip-lists HTML/UI routes, and **fails on any undocumented route**. The config request/response schema is built from `param_definitions.json` (SoT); response schemas for `/api/v1/last-run` and `/healthz` are read from their existing JSON Schema files and **inlined as components** (self-contained spec, internal `$ref`s). A pytest test regenerates and asserts equality with the committed file.

**Tech Stack:** Python 3 stdlib (`json`, `argparse`, `pathlib`), `emhass.web_server` (for `app.url_map`), `unittest` + `pytest`, `ruff`.

## Refinement vs spec (flagged): external schema handling

Spec Decision #6 says responses `$ref docs/api/v1/last-run.schema.json`. **This plan inlines** those schema files into `components.schemas` (reading them at generation time) and uses **internal** `$ref: "#/components/schemas/LastRun"`. Rationale: a relative external file `$ref` from `src/emhass/static/openapi.json` to `docs/api/...` is fragile (consumer must resolve cross-file paths); inlining yields a self-contained `openapi.json` while keeping the `.schema.json` files as the single authored source (the drift test regenerates, so editing a `.schema.json` without regenerating fails CI). Same DRY intent, better artifact.

```python
#!/usr/bin/env python3
"""Generate src/emhass/static/openapi.json from the EMHASS route map + schema sources.

Stdlib only. Run: python scripts/generate_openapi.py
"""

import argparse
import json
from pathlib import Path

_ATOMIC = {
    "string": {"type": "string"},
    "int": {"type": "integer"},
    "float": {"type": "number"},
    "boolean": {"type": "boolean"},
    "time": {"type": "string"},
    "object": {"type": "object"},
}


def _base_schema(base: str, param: dict) -> dict:
    if base == "select":
        return {"type": "string", "enum": list(param.get("select_options", []))}
    if base in _ATOMIC:
        return dict(_ATOMIC[base])
    raise SystemExit(f"generate_openapi: unknown input base type {base!r} (param input={param.get('input')!r})")


def _input_to_schema(input_str: str, param: dict) -> dict:
    """Map a param_definitions `input` string + param dict to an OpenAPI schema.

    For array.* inputs, the param `default_value` is the per-element scalar template,
    so it is placed on items.default; scalars get `default` directly. title/description/
    x-unit annotate the outermost property.
    """
    parts = input_str.split(".")
    leaf = _base_schema(parts[-1], param)
    dv = param.get("default_value", None)
    if dv is not None:
        leaf["default"] = dv
    schema = leaf
    for _ in parts[:-1]:          # one wrap per leading "array"
        schema = {"type": "array", "items": schema}
    if param.get("friendly_name"):
        schema["title"] = param["friendly_name"]
    desc = (param.get("Description") or "").strip()
    unit = param.get("unit")
    if unit and unit not in ("none", ""):
        schema["x-unit"] = unit
        desc = (desc + f" (unit: {unit})").strip()
    if desc:
        schema["description"] = desc
    return schema
```

```python
def build_config_component(param_defs: dict) -> dict:
    """Flatten param_definitions sections into a single OpenAPI object schema.

    Raises if a param key appears in more than one section (don't silently overwrite).
    """
    props: dict = {}
    for section, params in param_defs.items():
        for key, param in params.items():
            if key in props:
                raise SystemExit(f"generate_openapi: duplicate param key {key!r} across sections")
            props[key] = _input_to_schema(param["input"], param)
    return {"type": "object", "properties": props}
```

```python
def discovered_routes() -> set[tuple[str, str]]:
    """Return {(path, METHOD)} from the live Quart url_map. Normalises <x> -> {x}."""
    from emhass.web_server import app
    import re

    out: set[tuple[str, str]] = set()
    for rule in app.url_map.iter_rules():
        path = re.sub(r"<(?:[^:<>]+:)?([^<>]+)>", r"{\1}", rule.rule)
        for method in rule.methods or set():
            if method in ("HEAD", "OPTIONS"):
                continue
            out.add((path, method))
    return out


def assert_no_undocumented(routes: set, curated: dict, skip: set) -> None:
    for path, method in routes:
        if path in skip:
            continue
        if path not in curated or method not in curated[path]:
            raise SystemExit(
                f"generate_openapi: undocumented route {method} {path} — add to CURATED or SKIP"
            )
```

Note: Quart/Werkzeug auto-adds `HEAD`/`OPTIONS`; they are filtered. Confirm in your checkout that `iter_rules()` exists on the Quart `url_map` (Quart re-exports Werkzeug routing — it does on current versions).

```python
CURATED = {
    "/get-config": {"GET"},
    "/get-config/defaults": {"GET"},
    "/set-config": {"POST"},
    "/get-json": {"POST"},
    "/action/{action_name}": {"POST"},
    "/api/v1/last-run": {"GET"},
    "/healthz": {"GET"},
}

_PLAN_OUTPUT_DOC = "https://github.com/davidusb-geek/emhass/blob/master/docs/plan_output_schema.md"


def _load_json(path: Path) -> dict:
    data = json.loads(path.read_text(encoding="utf-8"))
    data.pop("$schema", None)   # not meaningful inside an openapi component
    data.pop("$id", None)
    return data


def _schema_version() -> str:
    from emhass.command_line import EMHASS_SCHEMA_VERSION
    return EMHASS_SCHEMA_VERSION


def build_spec() -> dict:
    param_defs = json.loads(_PARAM_DEFS.read_text(encoding="utf-8"))
    components = {
        "Config": build_config_component(param_defs),
        "LastRun": _load_json(_LAST_RUN_SCHEMA),
        "Healthz": _load_json(_HEALTHZ_SCHEMA),

            "201": {"description": "Config JSON", **json_ct(config_ref)}}}},

            "201": {"description": "Default config JSON", **json_ct(config_ref)}}}},
        "/set-config": {"post": {"summary": "Save config",

                          "500": {"description": "Conversion failure"}}}},

                        "content": {"application/json": {"schema": {"type": "object"}}},

                "400": {"description": "Action failure"}}}},
```
